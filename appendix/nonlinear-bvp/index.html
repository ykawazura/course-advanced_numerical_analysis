    非線形境界値問題

    この節の位置づけ

    第7回の有限差分法は、右辺 \(f\) が既知の線形な境界値問題を三重対角線形系に直して解いた。右辺が未知関数 \(y\) を含むと離散系そのものが非線形になり、多変数ニュートン法が要る。第2回（求根）・第3回（直接法）・第7回（有限差分）の道具が、ここで一つに組み合わさる。

    離散系が非線形になる

    右辺が \(-y''=f(x,y)\) のように未知関数 \(y\) を含む非線形問題では、離散化した連立方程式

    \[ \mathbf F(\mathbf y) = A\mathbf y - \mathbf f(\mathbf y) = \mathbf 0 \]

    が \(\mathbf y\) について非線形になる。これは多変数ニュートン法の出番である。各反復でヤコビアン \(J=\partial\mathbf F/\partial\mathbf y\) を組み、線形系 \(J\,\Delta\mathbf y=-\mathbf F\) を解く。ここで \(A\) は三重対角、\(\mathbf f(\mathbf y)\) の各成分 \(f_i\) は \(y_i\) のみに依存するので \(\partial f_i/\partial y_j\) は対角である。したがって \(J=A-\operatorname{diag}(\partial f_i/\partial y_i)\) も三重対角になり、各ニュートン反復の線形系をふたたびトーマス法で \(O(n)\) で解ける。第3回（直接法）・本補足（多変数ニュートン法）・第7回（有限差分）の道具がそのまま組み合わさる。

    例題2 への適用

    例題2 \(y''=\tfrac32y^2\), \(y(0)=4\), \(y(1)=1\)（厳密解 \(y=4/(1+x)^2\)）で具体的に書き下す。\(-y''=f(x,y)\) の形に直すと \(f(x,y)=-\tfrac32y^2\) である。例題1 と同じく両辺に \(h^2\) を掛けると、内点 \(i=1,\dots,N-1\) の残差は

    \[ F_i(\mathbf y) = -y_{i-1} + 2y_i - y_{i+1} + \tfrac32 h^2 y_i^2 = 0 \]

    となる（\(i=1\) と \(i=N-1\) では既知の境界値 \(y_0=4\)・\(y_N=1\) を右辺へ移項する）。\(F_i\) に現れる未知数は \(y_{i-1},y_i,y_{i+1}\) の3つだけなので、ヤコビアンは

    \[ J_{i,i\pm1} = \frac{\partial F_i}{\partial y_{i\pm1}} = -1, \qquad J_{ii} = \frac{\partial F_i}{\partial y_i} = 2 + 3h^2 y_i \]

    の三重対角になる。\((N-1)\times(N-1)\) 行列として書き下すと

    \[ J(\mathbf y) = \begin{pmatrix} 2+3h^2y_1 & -1 & & & \\ -1 & 2+3h^2y_2 & -1 & & \\ & \ddots & \ddots & \ddots & \\ & & -1 & 2+3h^2y_{N-2} & -1 \\ & & & -1 & 2+3h^2y_{N-1} \end{pmatrix} \]

    である。線形の場合の \(A\) と違って \(1/h^2\) が掛かっていないのは、残差を \(h^2\) 倍した形で定義したためで、\(h^2A=\operatorname{tridiag}(-1,2,-1)\) の対角に \(3h^2y_i\) が足された形になっている。\(\mathbf y\) に依存するのは対角だけなので、反復ごとに組み直すのは \(N-1\) 個の対角成分だけでよい（副対角は \(-1\) のまま変わらない）。\(y_i>0\) のあいだ \(\lvert J_{ii}\rvert>\lvert J_{i,i-1}\rvert+\lvert J_{i,i+1}\rvert\) が成り立つ。つまり対角優位なので、ピボット交換を持たないトーマス法をそのまま使える。初期値は境界を結ぶ直線 \(y^{(0)}_i=4-3x_i\) とし、各反復で \(J\,\Delta\mathbf y=-\mathbf F\) を解いて \(\mathbf y\leftarrow\mathbf y+\Delta\mathbf y\) と更新する。打ち切りは修正量 \(\lVert\Delta\mathbf y\rVert_\infty<10^{-12}\) とした。

    サンプルコード

    トーマス法 thomas は第7回の有限差分法から、引数も本体も変えずに写したものである。

    #include <stdio.h>
#include <stdlib.h>
#include <math.h>

/* トーマス法（三重対角ガウス消去, O(n)）
   a[i] x[i-1] + b[i] x[i] + c[i] x[i+1] = d[i],  i=0..n-1
   a[0], c[n-1] は使わない。cp, dp は作業配列。 */
void thomas(int n, const double a[], const double b[], const double c[],
            const double d[], double x[]) {
    double *cp = malloc(n * sizeof(double));
    double *dp = malloc(n * sizeof(double));
    cp[0] = c[0]/b[0];
    dp[0] = d[0]/b[0];
    for (int i=1;i<n;i++){               /* 前進消去 */
        double m = b[i] - a[i]*cp[i-1];
        cp[i] = c[i]/m;
        dp[i] = (d[i] - a[i]*dp[i-1])/m;
    }
    x[n-1] = dp[n-1];                    /* 後退代入 */
    for (int i=n-2;i>=0;i--) x[i] = dp[i] - cp[i]*x[i+1];
    free(cp);
    free(dp);
}

/* 例題2 の厳密解 y = 4/(1+x)^2 */
double exact2(double x) {
    return 4.0/((1.0+x)*(1.0+x));
}

/* 例題2: y'' = (3/2) y^2, y(0)=4, y(1)=1 を中心差分で離散化し、
   多変数ニュートン法で解く。内点 i=1..N-1 (m=N-1 個) の残差は、
   両辺に h^2 を掛けて
     F_i = -y_{i-1} + 2 y_i - y_{i+1} + (3/2) h^2 y_i^2
   （i=1 と i=N-1 では既知の境界値を右辺へ移項する）。
   ヤコビアンは三重対角で J_ii = 2 + 3 h^2 y_i, J_{i,i±1} = -1 となり、
   y_i>0 のあいだ対角優位なのでピボットなしのトーマス法で解ける。
   verbose なら反復ごとの残差を表示する。反復回数を *nit、
   2次の片側差分で推定した y'(0) を *yp0 に返し、最大誤差を返す。 */
double fd_newton(int N, int verbose, int *nit, double *yp0) {
    int m = N-1;
    double h = 1.0/N;
    double alpha = 4.0, beta = 1.0;              /* y(0)=4, y(1)=1 */
    double *a  = malloc(m * sizeof(double));
    double *b  = malloc(m * sizeof(double));
    double *c  = malloc(m * sizeof(double));
    double *d  = malloc(m * sizeof(double));
    double *y  = malloc(m * sizeof(double));
    double *F  = malloc(m * sizeof(double));
    double *dy = malloc(m * sizeof(double));
    for (int i=0;i<m;i++)                        /* 初期値は境界を結ぶ直線 */
        y[i] = alpha + (beta-alpha)*(i+1)*h;
    double emax = 0.0;
    int k = 0;
    while (k < 20) {
        double fmax = 0.0, dmax = 0.0;
        for (int i=0;i<m;i++){                   /* 残差とヤコビアンを組む */
            double left  = (i==0)   ? alpha : y[i-1];
            double right = (i==m-1) ? beta  : y[i+1];
            F[i] = -left + 2.0*y[i] - right + 1.5*h*h*y[i]*y[i];
            a[i] = -1.0; c[i] = -1.0;
            b[i] = 2.0 + 3.0*h*h*y[i];           /* dF_i/dy_i */
            d[i] = -F[i];
            if (fabs(F[i])>fmax) fmax = fabs(F[i]);
        }
        thomas(m, a, b, c, d, dy);               /* J dy = -F を解く */
        for (int i=0;i<m;i++){
            y[i] += dy[i];
            if (fabs(dy[i])>dmax) dmax = fabs(dy[i]);
        }
        k++;
        emax = 0.0;
        for (int i=0;i<m;i++){
            double e = fabs(y[i] - exact2((i+1)*h));
            if (e>emax) emax = e;
        }
        if (verbose)
            printf(" %3d   %.3e     %.3e    %.3e\n", k, fmax, dmax, emax);
        if (dmax < 1e-12) break;
    }
    *nit = k;
    *yp0 = (-3.0*alpha + 4.0*y[0] - y[1])/(2.0*h);   /* 2次の片側差分 */
    free(a); free(b); free(c); free(d); free(y); free(F); free(dy);
    return emax;
}

int main(void){
    int nit; double yp0;
    printf("=== ニュートン反復の収束（例題2, N=40）===\n");
    printf("   k   ||F||_inf     ||dy||_inf   最大誤差\n");
    printf("------------------------------------------------\n");
    fd_newton(40, 1, &nit, &yp0);
    printf("\n=== メッシュ細分による収束次数（2次精度の確認）===\n");
    printf("    N       h       最大誤差        誤差比   反復   y'(0)\n");
    printf("---------------------------------------------------------------\n");
    int Ns[] = {10,20,40,80,160,320};
    double prev=0;
    for (int j=0;j<6;j++){
        double e = fd_newton(Ns[j], 0, &nit, &yp0);
        printf(" %5d   %7.5f   %.6e", Ns[j], 1.0/Ns[j], e);
        if (j) printf("    %.3f", prev/e); else printf("      --  ");
        printf("    %2d   %9.6f\n", nit, yp0);
        prev = e;
    }
    printf("\n(2次精度なら誤差比は約 4、y'(0) は撃ち抜き法の s* = -8 に近づく)\n");
    return 0;
}

    コンパイル・実行:

    cc -O2 -Wall -o fd_newton fd_newton.c -lm
./fd_newton

    この出力から3つのことが読める。

    • ニュートン反復は2次収束する。修正量は \(7.17\times10^{-1}\to4.32\times10^{-2}\to1.54\times10^{-4}\to1.94\times10^{-9}\) と減り、\(\lVert\Delta\mathbf y^{(k+1)}\rVert_\infty\approx C\lVert\Delta\mathbf y^{(k)}\rVert_\infty^2\) の \(C\) は \(0.084\to0.083\to0.081\) とほぼ一定である。多変数ニュートン法の2次収束が、そのまま現れている。
    • 反復の誤差と離散化の誤差は別物である。最大誤差は3回目で \(2.98\times10^{-4}\) に落ちたあと、反復をいくら続けても動かない。ニュートン法が消すのは離散方程式の残差 \(\mathbf F\) だけで、離散化そのものの誤差 \(O(h^2)\) は格子を細かくしないと減らない。
    • 反復回数は格子の細かさによらない。\(N\) を 10 から 320 まで 32 倍にしても、収束までは 5 回のままである。一方、最大誤差の比は \(3.95\to3.99\to4.00\) と \(4\) へ近づき、線形の例題1 と同じ2次収束になる。

    表の最後の列は、数値解から2次の片側差分

    \[ y'(0) \approx \frac{-3y_0+4y_1-y_2}{2h} \]

    で初期傾きを見積もったものである。格子を細かくすると \(-7.731\to-7.9996\) と \(-8\) に近づく（\(-8\) との差は \(0.269\to0.0785\to\cdots\to0.000356\) で、これも2次で減る）。この値は撃ち抜き法が割線法で当てた \(s^*=-8.00000000\) と一致する。終端を狙って初期傾きを探す道と、区間全体を一度に解く道の、どちらから攻めても同じ解に行き着く。

    まとめ

    • 右辺が \(y\) を含む境界値問題は、離散化すると非線形の連立方程式 \(\mathbf F(\mathbf y)=\mathbf 0\) になる。これを多変数ニュートン法で解く。
    • ヤコビアン \(J=\partial\mathbf F/\partial\mathbf y\) は、\(f_i\) が \(y_i\) だけに依存するので三重対角のままである。各反復の線形系はトーマス法で \(O(n)\) で解ける。
    • 反復の誤差（残差 \(\mathbf F\)）と離散化の誤差 \(O(h^2)\) は別物である。ニュートン法が消すのは前者だけで、後者は格子を細かくしないと減らない。
    • 例題2 では、格子の細かさによらず5回のニュートン反復で収束し、数値解から測った \(y'(0)\) は撃ち抜き法の \(s^*=-8\) に2次で近づく。
